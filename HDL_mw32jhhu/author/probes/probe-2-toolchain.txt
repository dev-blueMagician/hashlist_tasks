cat /etc/debian_version; iverilog -V 2>&1 | head -1; verilator --version; g++ --version | head -1; make --version | head -1; python3 --version
id; nproc; df -h /app | tail -1; ls -la /app; ls -la /app/data 2>&1 | head
mkdir -p /tmp/t && printf '%s\n' 'module ctr #(parameter W = 8) (input wire clk, input wire rst_n, input wire en, output reg [W-1:0] q);' '  always @(posedge clk or negedge rst_n) if (!rst_n) q <= 0; else if (en) q <= q + 1;' 'endmodule' > /tmp/t/d.v && cat /tmp/t/d.v
printf '%s\n' '`timescale 1ns/1ps' 'module tb #(parameter W = 8); reg clk = 0, rst_n = 0, en = 0; wire [W-1:0] q; reg [63:0] nonce; integer seed, i, n; reg [W-1:0] m;' 'ctr #(.W(W)) dut (.clk(clk), .rst_n(rst_n), .en(en), .q(q)); always #5 clk = ~clk;' 'initial begin if (!$value$plusargs("nonce=%h", nonce)) nonce = 0; if (!$value$plusargs("seed=%d", seed)) seed = 1; m = 0; n = 0; #12 rst_n = 1;' 'for (i = 0; i < 200; i = i + 1) begin @(negedge clk); en = $random(seed); @(posedge clk); #1 if (en) m = m + 1; if (q !== m) n = n + 1; end' '$display("CHECK %h count %s seed=%0d", nonce, n == 0 ? "PASS" : "FAIL", seed); $display("DONE %h", nonce); $finish; end endmodule' > /tmp/t/tb.v && cat /tmp/t/tb.v
cd /tmp/t && iverilog -g2012 -o sim.vvp -s tb -Ptb.W=12 d.v tb.v && vvp -n sim.vvp +nonce=00c0ffee12345678 +seed=7
cd /tmp/t && time verilator --binary --timing -j 2 -Wno-fatal -Wno-lint -Wno-style --timescale 1ns/1ps --x-initial unique --x-assign unique --top-module tb -Mdir obj -o vsim -GW=12 d.v tb.v > build.log 2>&1; echo exit=$?; tail -3 build.log
cd /tmp/t && for s in 1 2 3; do ./obj/vsim +verilator+seed+$s +verilator+rand+reset+2 +nonce=00c0ffee12345678 +seed=7 | grep -E 'CHECK|DONE'; done
